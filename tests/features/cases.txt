from contextlib import contextmanager, suppress
@contextmanager
def tagged(name):
    print("<" + name + ">")
    yield name.upper()
    print("</" + name + ">")
with tagged("p") as t:
    print(t)
with suppress(KeyError):
    {}["x"]
